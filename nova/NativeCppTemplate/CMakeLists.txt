cmake_minimum_required(VERSION 3.18)
project(NovaNativeComponent LANGUAGES CXX)

set(NOVA_SOURCES src/NovaComponent.cpp)
if(ANDROID)
  list(APPEND NOVA_SOURCES src/NovaJniBridge.cpp)
endif()
add_library(NovaComponent SHARED ${NOVA_SOURCES})
target_include_directories(NovaComponent PRIVATE include)
target_compile_features(NovaComponent PRIVATE cxx_std_17)
target_compile_options(NovaComponent PRIVATE -Wall -Wextra -fvisibility=hidden)
set_target_properties(NovaComponent PROPERTIES OUTPUT_NAME "NovaComponent")

option(NOVA_BUILD_TESTS "Build host-side native ABI tests" ON)
if(NOVA_BUILD_TESTS AND NOT ANDROID)
  enable_testing()
  add_executable(nova_native_tests tests/test_native.cpp src/NovaComponent.cpp)
  target_include_directories(nova_native_tests PRIVATE include)
  target_compile_features(nova_native_tests PRIVATE cxx_std_17)
  add_test(NAME nova_native_lifecycle COMMAND nova_native_tests)
endif()
